def server(input: Inputs, output: Outputs, session: Session):

    # Ex: Garrick's conf app that has many columns of cards and you can drag cards between columns. You need to know when column updates are made
    # Have the UI be within the app file, requires to author to program the UI logic. This is NOT trivial!!

    # Must draw each card
    for item in app_card_data():
        @output(id=item.id)
        @render.garrick_card
        def _():
            return {"title": "Hello Shiny React!", "content": input.txtin()}

    col_infos <- []
    for i in range(1, 4):
        col_data <- reactive({
            # Get the data for this column, e.g. from a database or in-memory store
            get_column_data(app_card_data(), i)
        })
        col_infos.append(col_data)

        @output(id=f"column_{i}_cards")
        @render.ui
        def _():
            cards = [ui.output_card(card_id) for card_id in col_data()]
            return cards



    @reactive.effect(input.card_dragged)
    def _():
        # When a card is dragged, update the data store with the new column assignment
        card_id, new_position = input.card_dragged()
        update_card_column(app_card_data, card_id, new_position)


    @reactive.calc
    def greeting():
        name = input.name()
        if not name:
            return "World"
        return name

    @shinyjson.render
    def txtout_title():
        return f"Hello, {greeting()}!"

    @shinyjson.render
    def txtout_count():
        return input.click_count()
